```ts
import type {Locale, Localized, Resolved} from '@/types/common';

function hasValue(value: unknown): boolean {
  if (value === undefined || value === null) return false;
  if (typeof value === 'string') return value.trim() !== '';
  if (Array.isArray(value)) return value.length > 0;
  return true;
}

export function isLocalized(value: unknown): value is Localized<unknown> {
  if (value === null || typeof value !== 'object' || Array.isArray(value)) return false;
  const keys = Object.keys(value);
  return keys.includes('en') && keys.every((key) => key === 'en' || key === 'ko');
}

/** Picks the value for `locale`, falling back to English when it is missing or blank. */
export function localize<T>(value: Localized<T>, locale: Locale): T {
  const preferred = value[locale];
  if (hasValue(preferred)) return preferred as T;
  if (locale !== 'en' && process.env.NODE_ENV === 'development') {
    console.warn(`[i18n] Missing "${locale}" value, falling back to English:`, value.en);
  }
  return value.en;
}

/** Deeply replaces every Localized field with its value for `locale`. */
export function resolveLocalized<T>(value: T, locale: Locale): Resolved<T> {
  if (Array.isArray(value)) {
    return value.map((item) => resolveLocalized(item, locale)) as Resolved<T>;
  }
  if (isLocalized(value)) {
    return resolveLocalized(localize(value, locale), locale) as Resolved<T>;
  }
  if (value !== null && typeof value === 'object') {
    return Object.fromEntries(
      Object.entries(value).map(([key, child]) => [key, resolveLocalized(child, locale)]),
    ) as Resolved<T>;
  }
  return value as Resolved<T>;
}
```

```ts
import type {PortableTextBlock} from '@portabletext/types';
import type {Localized, RichText} from '@/types/common';

/**
 * Builds Portable Text from a tiny markdown subset so mock content stays readable.
 * Blocks are separated by a blank line. Supported: paragraphs, "## " (h2), "### " (h3),
 * and blocks whose every line starts with "- " (bullet items).
 */
export function pt(source: string, keyPrefix = 'b'): RichText {
  const blocks: PortableTextBlock[] = [];
  let index = 0;

  const block = (text: string, style: string, listItem?: 'bullet'): PortableTextBlock => {
    const key = `${keyPrefix}${index++}`;
    return {
      _type: 'block',
      _key: key,
      style,
      markDefs: [],
      children: [{_type: 'span', _key: `${key}s`, text, marks: []}],
      ...(listItem ? {listItem, level: 1} : {}),
    };
  };

  for (const chunk of source.trim().split(/\n\s*\n/)) {
    const lines = chunk
      .split('\n')
      .map((line) => line.trim())
      .filter(Boolean);
    if (lines.length === 0) continue;

    if (lines.every((line) => line.startsWith('- '))) {
      for (const line of lines) blocks.push(block(line.slice(2), 'normal', 'bullet'));
    } else if (lines[0].startsWith('### ')) {
      blocks.push(block(lines.join(' ').slice(4), 'h3'));
    } else if (lines[0].startsWith('## ')) {
      blocks.push(block(lines.join(' ').slice(3), 'h2'));
    } else {
      blocks.push(block(lines.join(' '), 'normal'));
    }
  }

  return blocks;
}

export function localizedPt(value: {en: string; ko: string}): Localized<RichText> {
  return {en: pt(value.en), ko: pt(value.ko)};
}
```

```ts
import type {EventItem, EventStatus, TimelineYear} from '@/types/event';

/** An event stays "upcoming" until it has ended, so ongoing events still show registration. */
export function getEventStatus(event: {endsAt: string}, now: Date): EventStatus {
  return Date.parse(event.endsAt) >= now.getTime() ? 'upcoming' : 'past';
}

export function compareByStartAsc(a: {startsAt: string}, b: {startsAt: string}): number {
  return Date.parse(a.startsAt) - Date.parse(b.startsAt);
}

export function sortEventItems<T extends {status: EventStatus; startsAt: string}>(items: T[]): T[] {
  return [...items].sort((a, b) => {
    if (a.status !== b.status) return a.status === 'upcoming' ? -1 : 1;
    return a.status === 'upcoming' ? compareByStartAsc(a, b) : compareByStartAsc(b, a);
  });
}

export function buildTimeline(events: EventItem[]): TimelineYear[] {
  const byYear = new Map<number, TimelineYear['items']>();
  for (const event of [...events].sort(compareByStartAsc)) {
    // The ISO string carries Toronto local time, so its first four characters are the local year.
    const year = Number(event.startsAt.slice(0, 4));
    const items = byYear.get(year) ?? [];
    items.push({slug: event.slug, title: event.title, startsAt: event.startsAt, status: event.status, kind: event.kind});
    byYear.set(year, items);
  }
  return [...byYear.entries()].sort(([a], [b]) => a - b).map(([year, items]) => ({year, items}));
}
```

```ts
export function normalizeSearch(value: string): string {
  return value.normalize('NFC').toLocaleLowerCase().replace(/\s+/g, ' ').trim();
}

/** Matches the query against the name in both locales; spaces are optional (e.g. "김 지원" finds "김지원"). */
export function filterMentors<T extends {name: string; nameAlt: string}>(mentors: T[], query: string): T[] {
  const q = normalizeSearch(query);
  if (!q) return mentors;
  const compact = q.replace(/ /g, '');
  return mentors.filter((mentor) =>
    [mentor.name, mentor.nameAlt].some((name) => {
      const value = normalizeSearch(name);
      return value.includes(q) || value.replace(/ /g, '').includes(compact);
    }),
  );
}
```

```ts
import {isLocalized} from '@/lib/i18n/localize';

export type Found<T> = {path: string; value: T};

/** Visits every object and array inside `root`, depth-first. */
export function walk(root: unknown, visit: (value: unknown, path: string) => void, path = '$'): void {
  visit(root, path);
  if (Array.isArray(root)) {
    root.forEach((child, i) => walk(child, visit, `${path}[${i}]`));
  } else if (root !== null && typeof root === 'object') {
    for (const [key, child] of Object.entries(root)) walk(child, visit, `${path}.${key}`);
  }
}

type ImageLike = {url: string; width: number; height: number; alt: {en: string; ko?: string}};

export function findImages(root: unknown): Found<ImageLike>[] {
  const found: Found<ImageLike>[] = [];
  walk(root, (value, path) => {
    if (value && typeof value === 'object' && 'url' in value && 'width' in value && 'height' in value) {
      found.push({path, value: value as ImageLike});
    }
  });
  return found;
}

export function findLocalized(root: unknown): Found<{en: unknown; ko?: unknown}>[] {
  const found: Found<{en: unknown; ko?: unknown}>[] = [];
  walk(root, (value, path) => {
    if (isLocalized(value)) found.push({path, value});
  });
  return found;
}

export function isBlank(value: unknown): boolean {
  if (typeof value === 'string') return value.trim() === '';
  if (Array.isArray(value)) return value.length === 0;
  return value === undefined || value === null;
}
```

Append to `lib/i18n/paths.ts`:
```ts
/** Every static (non-slug) route of the public site, locale-less. */
export const STATIC_PATHS = [
  '/',
  '/about',
  '/about/reports',
  '/get-involved/mentorship',
  '/get-involved/mentors',
  '/get-involved/donate',
  '/events-news',
  '/contact',
] as const;

/** True for '#anchor', a static route, or /events-news/<known slug> — with optional ?query and #hash. */
export function isKnownInternalHref(href: string, contentSlugs: readonly string[]): boolean {
  if (href.startsWith('#')) return href.length > 1;
  const pathname = href.split(/[?#]/)[0];
  if ((STATIC_PATHS as readonly string[]).includes(pathname)) return true;
  const match = /^\/events-news\/([a-z0-9-]+)$/.exec(pathname);
  return match !== null && contentSlugs.includes(match[1]);
}
```

```ts
import 'server-only';
import {mentors} from '@/data/mock/mentors';
import {localize, resolveLocalized} from '@/lib/i18n/localize';
import type {Locale} from '@/types/common';
import type {Mentor} from '@/types/mentor';

const otherLocale = (locale: Locale): Locale => (locale === 'en' ? 'ko' : 'en');

export async function getMentors(locale: Locale): Promise<Mentor[]> {
  'use cache';
  return [...mentors]
    .sort((a, b) => a.order - b.order)
    .map((doc) => ({...resolveLocalized(doc, locale), nameAlt: localize(doc.name, otherLocale(locale))}));
}

export async function getFeaturedMentors(locale: Locale, limit = 4): Promise<Mentor[]> {
  'use cache';
  return (await getMentors(locale)).filter((mentor) => mentor.featured).slice(0, limit);
}
```

```ts
export async function getEvents(options: {
  locale: Locale;
  status?: EventStatus;
  kind?: EventKind;
  now?: string;

  const current = now ? new Date(now) : new Date();
  const items = events
    .map((doc) => toEventItem(doc, locale, current))
    .filter((event) => (!status || event.status === status) && (!kind || event.kind === kind));

}

  'use cache';
```

```ts
import 'server-only';
import {events} from '@/data/mock/events';
import {posts} from '@/data/mock/posts';
import {resolveLocalized} from '@/lib/i18n/localize';
import type {Locale} from '@/types/common';
import type {ContentItem} from '@/types/event';
import {getEvents, toEventItem} from './events';
import {categoryOf} from './helpers/content';
import {getPosts} from './posts';

/** Events (upcoming first, then past) followed by posts (newest first). */
export async function getAllContent(locale: Locale, now?: string): Promise<ContentItem[]> {
  'use cache';
  const [eventItems, postItems] = await Promise.all([getEvents({locale, now}), getPosts({locale})]);
  return [
    ...eventItems.map((event) => ({...event, type: 'event' as const})),
    ...postItems.map((post) => ({...post, type: 'post' as const})),
  ];
}

export async function getContentBySlug(locale: Locale, slug: string, now?: string): Promise<ContentItem | null> {
  'use cache';
  const event = events.find((doc) => doc.slug === slug);
  if (event) return {...toEventItem(event, locale, now ? new Date(now) : new Date()), type: 'event'};
  const post = posts.find((doc) => doc.slug === slug);
  if (post) return {...resolveLocalized(post, locale), type: 'post'};
  return null;
}

export async function getAllContentSlugs(): Promise<string[]> {
  return [...events, ...posts].map((doc) => doc.slug);
}

export async function getLatestHighlights(locale: Locale, limit = 3, now?: string): Promise<ContentItem[]> {
  'use cache';
  const all = await getAllContent(locale, now);
  const upcoming = all.filter((item) => item.type === 'event' && item.status === 'upcoming');
  const latestPosts = all.filter((item) => item.type === 'post');
  return [...upcoming, ...latestPosts].slice(0, limit);
}

export async function getRelated(locale: Locale, slug: string, limit = 3, now?: string): Promise<ContentItem[]> {
  'use cache';
  const item = await getContentBySlug(locale, slug, now);
  if (!item) return [];
  const category = categoryOf(item);
  const all = await getAllContent(locale, now);
  return all.filter((other) => other.slug !== slug && categoryOf(other) === category).slice(0, limit);
}

export async function getFeaturedContent(locale: Locale, now?: string): Promise<ContentItem | null> {
  'use cache';
  const all = await getAllContent(locale, now);
  return all.find((item) => item.featured) ?? all[0] ?? null;
}
```
